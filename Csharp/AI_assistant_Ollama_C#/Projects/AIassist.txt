Мені потрібно згенерувати консольну програму мовою C#, яка буде генерувати текст з допомогою локальної Ollama phi3.

```
using System.Net.Http.Json;
using System.Text.Json.Serialization;

const string OllamaUrl = "http://localhost:11434/api/generate";
const string Model = "phi3";

using var httpClient = new HttpClient
{
    Timeout = TimeSpan.FromMinutes(5)
};

Console.OutputEncoding = System.Text.Encoding.UTF8;

Console.WriteLine("========================================");
Console.WriteLine("        C# Ollama AI Assistant");
Console.WriteLine("========================================");
Console.WriteLine();
Console.WriteLine($"Модель: {Model}");
Console.WriteLine("Для виходу введіть: exit");
Console.WriteLine();

while (true)
{
    Console.Write("You: ");

    string? prompt = Console.ReadLine();

    if (string.IsNullOrWhiteSpace(prompt))
    {
        continue;
    }

    if (prompt.Equals("exit", StringComparison.OrdinalIgnoreCase))
    {
        break;
    }

    try
    {
        Console.WriteLine();
        Console.Write("AI: ");

        var request = new OllamaRequest
        {
            Model = Model,
            Prompt = prompt,
            Stream = false
        };

        var response = await httpClient.PostAsJsonAsync(
            OllamaUrl,
            request);

        response.EnsureSuccessStatusCode();

        var result = await response.Content
            .ReadFromJsonAsync<OllamaResponse>();

        if (result == null || string.IsNullOrWhiteSpace(result.Response))
        {
            Console.WriteLine("Модель не повернула відповідь.");
        }
        else
        {
            Console.WriteLine(result.Response);
        }
    }
    catch (HttpRequestException)
    {
        Console.WriteLine();
        Console.WriteLine("Помилка: не вдалося підключитися до Ollama.");
        Console.WriteLine("Переконайтеся, що Ollama запущена.");
        Console.WriteLine("Перевірте адресу: http://localhost:11434");
    }
    catch (TaskCanceledException)
    {
        Console.WriteLine();
        Console.WriteLine("Запит перевищив допустимий час очікування.");
    }
    catch (Exception ex)
    {
        Console.WriteLine();
        Console.WriteLine($"Помилка: {ex.Message}");
    }

    Console.WriteLine();
}

Console.WriteLine("Програму завершено.");

public class OllamaRequest
{
    [JsonPropertyName("model")]
    public string Model { get; set; } = string.Empty;

    [JsonPropertyName("prompt")]
    public string Prompt { get; set; } = string.Empty;

    [JsonPropertyName("stream")]
    public bool Stream { get; set; }
}

public class OllamaResponse
{
    [JsonPropertyName("response")]
    public string Response { get; set; } = string.Empty;

    [JsonPropertyName("done")]
    public bool Done { get; set; }
}
```
